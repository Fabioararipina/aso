const { Pool } = require('pg');
require('dotenv').config();
const pool = new Pool({
  user: process.env.DB_USER,
  host: process.env.DB_HOST,
  database: process.env.DB_NAME,
  password: process.env.DB_PASSWORD,
  port: process.env.DB_PORT
});
async function run() {
  try {
    console.log('🔧 Corrigindo estrutura da tabela agendamentos...');
    await pool.query("ALTER TABLE agendamentos ALTER COLUMN paciente_telefone DROP NOT NULL;");
    await pool.query("ALTER TABLE agendamentos ADD COLUMN IF NOT EXISTS paciente_id INTEGER REFERENCES pacientes(id);");
    await pool.query("ALTER TABLE agendamentos ADD COLUMN IF NOT EXISTS medico_id INTEGER REFERENCES medicos(id);");
    await pool.query("ALTER TABLE agendamentos ADD COLUMN IF NOT EXISTS valor NUMERIC(10,2);");
    console.log('✅ Banco de dados corrigido com sucesso!');
    process.exit(0);
  } catch (err) {
    console.error('❌ Erro ao corrigir banco:', err);
    process.exit(1);
  }
}
run();
EOF
